function imprimirPares() {
    let i = 2;
    let mensagem = '';

    do{
        mensagem += i + "\n";
        i += 2; // i = i + 2
    } while (i <= 20);
    return mensagem;
}

function principal() {
    alert ('Numeros pares de 2 a 20: \n' + imprimirPares());
}

principal();
